// ==============================================================================
// San Andreas Pathfinder 3D - SA-MP Shortcut & Trajectory Recorder
// ==============================================================================
// Hardened CLEO script for SA-MP, using the native IniFiles.cleo plugin
// Records continuous 3D trajectories (jumps, curves, shortcuts) into cleo/shortcuts.ini
// No open/close file opcodes, no null pointers, no crashes
//
// CONTROLS:
//   Key 'I' or Ctrl + 1 : START recording a trajectory (REC button)
//   Key 'O' or Ctrl + 2 : STOP and save the trajectory (STOP button)
//
// OUTPUT FORMAT (cleo/shortcuts.ini):
//   Section [id_pt] with x, y, z keys, and section [id] with the point total
//   Continuous auto-sampling every 10.0 metres while moving or falling
// ==============================================================================

{$CLEO .cs}
Alloc($PLAYER_ACTOR, 3)

script_name "SRECOR"

// Formal local variable type declarations for Sanny Builder
var
    0@ : Float  // Current X coordinate
    1@ : Float  // Current Y coordinate
    2@ : Float  // Current Z coordinate
    10@ : Int   // Recording state: 0 = idle, 1 = recording
    11@ : Int   // Counter / ID of the current trajectory (Shortcut #1, #2, ...)
    12@ : Int   // Point counter within the current trajectory
    13@ : Float // Last saved X
    14@ : Float // Last saved Y
    15@ : Float // Last saved Z
    16@ : Float // Squared travelled distance (dist_sq)
    17@ : Float // Math scratch variable
end

// Clean initialisation
10@ = 0 // recording_active = 0
11@ = 0 // shortcut_id = 0
12@ = 0 // point_index = 0

// Initial passive wait so SA-MP loads cleanly
wait 1500

// =============================================================================
// MAIN LOOP
// =============================================================================
:MAIN_LOOP
wait 10

// 1. Detect START key: 'I' (73) or Ctrl + 1
:CHECK_KEY_START
if
    0AB0: key_pressed 73  // Key 'I'
then
    goto @START_RECORDING
end

if and
    0AB0: key_pressed 17  // Ctrl
    0AB0: key_pressed 49  // 1
then
    goto @START_RECORDING
end

if and
    0AB0: key_pressed 17  // Ctrl
    0AB0: key_pressed 97  // Numpad 1
then
    goto @START_RECORDING
end

goto @CHECK_KEY_STOP

// Start recording a trajectory
:START_RECORDING
if
    10@ == 1
then
    // Already recording, ignore the repeat
    goto @CHECK_KEY_STOP
end

if
    0256: player 0 defined
jf @CHECK_KEY_STOP

// Enable recording and prime the counters
10@ = 1
11@ += 1  // New shortcut number (#1, #2, #3...)
12@ = 1   // First point (#1)

// Capture the initial position, on foot or in a vehicle
00A0: store_actor $PLAYER_ACTOR position_to 0@ 1@ 2@
0087: 13@ = 0@
0087: 14@ = 1@
0087: 15@ = 2@

// Write point #1 to cleo/shortcuts.ini via IniFiles.cleo
0AD3: 20@v = string_format "%d_%d" 11@ 12@
0AF3: write_float 0@ to_ini_file "cleo/shortcuts.ini" section 20@v key "x"
0AF3: write_float 1@ to_ini_file "cleo/shortcuts.ini" section 20@v key "y"
0AF3: write_float 2@ to_ini_file "cleo/shortcuts.ini" section 20@v key "z"

// Save the shortcut metadata
0AD3: 20@v = string_format "%d" 11@
0AF1: write_int 12@ to_ini_file "cleo/shortcuts.ini" section 20@v key "points"
0AF1: write_int 11@ to_ini_file "cleo/shortcuts.ini" section "meta" key "total_shortcuts"

// Audiovisual feedback
018C: play_sound 1057 at 0.0 0.0 0.0  // Positive beep
0ACD: show_text_highpriority "REC: ~g~GRABANDO TRAYECTORIA...~w~" time 2000
wait 300 // Debounce

// -----------------------------------------------------------------------------
// 2. Detect STOP-AND-SAVE key: 'O' (79) or Ctrl + 2
// -----------------------------------------------------------------------------
:CHECK_KEY_STOP
if
    0AB0: key_pressed 79  // Key 'O'
then
    goto @STOP_RECORDING
end

if and
    0AB0: key_pressed 17  // Ctrl
    0AB0: key_pressed 50  // 2
then
    goto @STOP_RECORDING
end

if and
    0AB0: key_pressed 17  // Ctrl
    0AB0: key_pressed 98  // Numpad 2
then
    goto @STOP_RECORDING
end

goto @SAMPLE_TRAJECTORY

// Stop recording the trajectory
:STOP_RECORDING
if
    10@ == 0
then
    // Not recording: tell the user
    018C: play_sound 1058 at 0.0 0.0 0.0
    0ACD: show_text_highpriority "~r~SIN GRABACION ACTIVA~w~: Presiona Tecla I para iniciar" time 2500
    wait 300
    goto @SAMPLE_TRAJECTORY
end

if
    0256: player 0 defined
jf @FINALIZE_STOP

// Capture the final position
00A0: store_actor $PLAYER_ACTOR position_to 0@ 1@ 2@

// Squared distance: dist_sq = dx^2 + dy^2 + dz^2
0087: 16@ = 0@
0063: 16@ -= 13@
006B: 16@ *= 16@ // dx^2

0087: 17@ = 1@
0063: 17@ -= 14@
006B: 17@ *= 17@ // dy^2

005B: 16@ += 17@ // dx^2 + dy^2

0087: 17@ = 2@
0063: 17@ -= 15@
006B: 17@ *= 17@ // dz^2

005B: 16@ += 17@ // dist_sq

// If we moved more than 0.5 m (0.5^2 = 0.25) since the last point, log the exact final point
if
    16@ >= 0.25
then
    12@ += 1
    0AD3: 20@v = string_format "%d_%d" 11@ 12@
    0AF3: write_float 0@ to_ini_file "cleo/shortcuts.ini" section 20@v key "x"
    0AF3: write_float 1@ to_ini_file "cleo/shortcuts.ini" section 20@v key "y"
    0AF3: write_float 2@ to_ini_file "cleo/shortcuts.ini" section 20@v key "z"
end

// Update the shortcut's final point count
0AD3: 20@v = string_format "%d" 11@
0AF1: write_int 12@ to_ini_file "cleo/shortcuts.ini" section 20@v key "points"

:FINALIZE_STOP
10@ = 0 // Stop recording

// Audiovisual feedback
018C: play_sound 1058 at 0.0 0.0 0.0  // Stop beep
0ACD: show_text_highpriority "Atajo ~g~GUARDADO~w~ en cleo/shortcuts.ini" time 3000
wait 300 // Debounce

// -----------------------------------------------------------------------------
// 3. Background auto-sampling (every 10.0 metres travelled or fallen)
// -----------------------------------------------------------------------------
:SAMPLE_TRAJECTORY
if
    10@ == 1
jf @MAIN_LOOP

if
    0256: player 0 defined
jf @MAIN_LOOP

// Read the current position
00A0: store_actor $PLAYER_ACTOR position_to 0@ 1@ 2@

// Squared distance: dist_sq = dx^2 + dy^2 + dz^2
0087: 16@ = 0@
0063: 16@ -= 13@
006B: 16@ *= 16@ // dx^2

0087: 17@ = 1@
0063: 17@ -= 14@
006B: 17@ *= 17@ // dy^2

005B: 16@ += 17@ // dx^2 + dy^2

0087: 17@ = 2@
0063: 17@ -= 15@
006B: 17@ *= 17@ // dz^2

005B: 16@ += 17@ // dist_sq

// Fire once we have travelled or fallen at least 10.0 m (10.0^2 = 100.0), matching the GTA:SA standard:
if
    16@ >= 100.0
jf @MAIN_LOOP

// Log a new point in shortcuts.ini
12@ += 1
0AD3: 20@v = string_format "%d_%d" 11@ 12@
0AF3: write_float 0@ to_ini_file "cleo/shortcuts.ini" section 20@v key "x"
0AF3: write_float 1@ to_ini_file "cleo/shortcuts.ini" section 20@v key "y"
0AF3: write_float 2@ to_ini_file "cleo/shortcuts.ini" section 20@v key "z"

// Update this shortcut's running point count
0AD3: 20@v = string_format "%d" 11@
0AF1: write_int 12@ to_ini_file "cleo/shortcuts.ini" section 20@v key "points"

// Update the last-point coordinates
0087: 13@ = 0@
0087: 14@ = 1@
0087: 15@ = 2@

// Shutter click so you know a point was logged while driving
018C: play_sound 1137 at 0.0 0.0 0.0

goto @MAIN_LOOP
